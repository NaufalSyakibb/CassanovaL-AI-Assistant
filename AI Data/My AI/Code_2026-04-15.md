CREATE DATABASE IF NOT EXISTS db_kampus;
USE db_kampus;

-- Tabel Mahasiswa
CREATE TABLE IF NOT EXISTS tb_mahasiswa (
    nim         VARCHAR(13)  PRIMARY KEY,
    nama        VARCHAR(50)  NOT NULL,
    alamat      VARCHAR(100),
    jenis_kelamin CHAR(1),   -- 'L' atau 'P'
    tgl_lahir   DATE,
    ipk         DECIMAL(3,2)
);

-- Tabel Mata Kuliah
CREATE TABLE IF NOT EXISTS tb_mk (
    kode_mk     VARCHAR(7)   PRIMARY KEY,
    nama_mk     VARCHAR(50)  NOT NULL,
    sks         INT,
    semester    INT,
    kode_dsn    VARCHAR(6)
);

-- Tabel Dosen
CREATE TABLE IF NOT EXISTS tb_dosen (
    kode_dsn    VARCHAR(6)   PRIMARY KEY,
    nama_dsn    VARCHAR(50)  NOT NULL,
    prodi       VARCHAR(30)
);

-- Tabel Nilai
CREATE TABLE IF NOT EXISTS tb_nilai (
    nim         VARCHAR(13),
    kode_mk     VARCHAR(7),
    quiz        DECIMAL(5,2),
    uts         DECIMAL(5,2),
    uas         DECIMAL(5,2),
    PRIMARY KEY (nim, kode_mk),
    FOREIGN KEY (nim)     REFERENCES tb_mahasiswa(nim),
    FOREIGN KEY (kode_mk) REFERENCES tb_mk(kode_mk)
);

-- Mengelompokkan data dan memfilter hasil grup

-- --------------------------------------------------
-- 6.1 GROUP BY dasar
-- --------------------------------------------------

-- Hitung jumlah mahasiswa per jenis kelamin
SELECT   jenis_kelamin,
         COUNT(*) AS jumlah_mahasiswa
FROM     tb_mahasiswa
GROUP BY jenis_kelamin;

-- --------------------------------------------------
-- 6.2 Fungsi agregat dengan GROUP BY
-- --------------------------------------------------

-- Nilai rata-rata, max, min per mata kuliah
SELECT   kode_mk,
         COUNT(*)      AS jumlah_peserta,
         AVG(uas)      AS rata_uas,
         MAX(uas)      AS nilai_uas_tertinggi,
         MIN(uas)      AS nilai_uas_terendah,
         SUM(uas)      AS total_uas
FROM     tb_nilai
GROUP BY kode_mk;

-- --------------------------------------------------
-- 6.3 HAVING — Filter pada hasil GROUP BY
-- HAVING berbeda dari WHERE: HAVING filter SETELAH pengelompokan
-- --------------------------------------------------

-- Tampilkan hanya mk yang rata-rata UAS-nya >= 75
SELECT   kode_mk,
         AVG(uas) AS rata_uas
FROM     tb_nilai
GROUP BY kode_mk
HAVING   AVG(uas) >= 75;

-- Jumlah peserta > 1 dan rata-rata UAS > 80
SELECT   kode_mk,
         COUNT(*) AS peserta,
         AVG(uas) AS rata_uas
FROM     tb_nilai
GROUP BY kode_mk
HAVING   COUNT(*) > 1
  AND    AVG(uas) > 80;

-- --------------------------------------------------
-- 6.4 WHERE vs HAVING
-- --------------------------------------------------

-- WHERE dievaluasi SEBELUM GROUP BY
-- HAVING dievaluasi SETELAH GROUP BY

-- Filter mahasiswa tertentu dulu, baru kelompokkan
SELECT   kode_mk,
         AVG(uas) AS rata_uas
FROM     tb_nilai
WHERE    uas > 60           -- filter baris dulu
GROUP BY kode_mk
HAVING   AVG(uas) >= 75;   -- filter grup kemudian

-- --------------------------------------------------
-- 6.5 GROUP BY dengan ORDER BY
-- --------------------------------------------------

FROM     tb_nilai

-- 8.1 Rekap nilai mahasiswa lengkap

SELECT
    m.nim,
    m.nama,
    mk.nama_mk,
    n.quiz,
    n.uts,
    n.uas,
    ROUND((n.quiz * 0.20 + n.uts * 0.30 + n.uas * 0.50), 2) AS nilai_akhir,
    CASE
        WHEN (n.quiz*0.2 + n.uts*0.3 + n.uas*0.5) >= 85 THEN 'A'
        WHEN (n.quiz*0.2 + n.uts*0.3 + n.uas*0.5) >= 75 THEN 'B'
        WHEN (n.quiz*0.2 + n.uts*0.3 + n.uas*0.5) >= 65 THEN 'C'
        WHEN (n.quiz*0.2 + n.uts*0.3 + n.uas*0.5) >= 55 THEN 'D'
        ELSE 'E'
    END AS grade
FROM       tb_nilai    n
JOIN       tb_mahasiswa m  USING (nim)
JOIN       tb_mk        mk USING (kode_mk)
ORDER BY   mk.nama_mk, nilai_akhir DESC;

-- --------------------------------------------------
-- 8.2 Statistik per mata kuliah + filter HAVING
-- --------------------------------------------------

SELECT
    mk.kode_mk,
    mk.nama_mk,
    mk.sks,
    COUNT(n.nim)    AS peserta,
    ROUND(AVG(n.uas), 2) AS avg_uas,
    MAX(n.uas)      AS max_uas,
    MIN(n.uas)      AS min_uas
FROM     tb_mk mk
LEFT JOIN tb_nilai n USING (kode_mk)
GROUP BY mk.kode_mk, mk.nama_mk, mk.sks
HAVING   peserta > 0
ORDER BY avg_uas DESC
LIMIT    3;

-- --------------------------------------------------
-- 8.3 Mahasiswa dengan nilai akhir di atas rata-rata
-- --------------------------------------------------

SELECT
    m.nama,
    mk.nama_mk,
    ROUND((n.quiz*0.2 + n.uts*0.3 + n.uas*0.5), 2) AS nilai_akhir
FROM  tb_nilai n
JOIN  tb_mahasiswa m  USING (nim)
JOIN  tb_mk        mk USING (kode_mk)
WHERE (n.quiz*0.2 + n.uts*0.3 + n.uas*0.5) >
      (SELECT AVG(quiz*0.2 + uts*0.3 + uas*0.5) FROM tb_nilai)

bagaimana cara menentukan nilai tertinggi / max() dari semua mahasiswa yang ada

**Coding:** **[Cara menentukan nilai tertinggi (MAX) dari semua mahasiswa]**

Kamu bisa menggunakan fungsi agregat `MAX()` pada kolom `uas`, `uts`, `quiz`, atau **nilai akhir** (gabungan dari ketiganya) untuk menemukan nilai tertinggi di seluruh tabel `tb_nilai`. Jika kamu ingin mencari nilai tertinggi **per mahasiswa**, gunakan `GROUP BY nim`.

---

### **1. Nilai UAS Tertinggi dari Semua Mahasiswa**
Jika kamu hanya ingin nilai **UAS tertinggi** dari seluruh data (tanpa memperhatikan mata kuliah):

```sql
SELECT MAX(uas) AS nilai_uas_tertinggi
FROM tb_nilai;
```

> 🔥 **Pro Tip:**
> Jika kamu juga ingin melihat **siapa mahasiswa yang mendapat nilai tertinggi**, gunakan query berikut:

```sql
SELECT
    m.nim,
    m.nama,
    n.kode_mk,
    mk.nama_mk,
    n.uas AS nilai_uas
FROM tb_nilai n
JOIN tb_mahasiswa m USING (nim)
JOIN tb_mk mk USING (kode_mk)
WHERE n.uas = (SELECT MAX(uas) FROM tb_nilai);
```

---

### **2. Nilai Akhir Tertinggi (Gabungan Quiz, UTS, UAS)**
Jika kamu ingin mencari **nilai akhir tertinggi** (dihitung dari `quiz * 0.2 + uts * 0.3 + uas * 0.5`):

```sql
SELECT
    m.nim,
    m.nama,
    mk.nama_mk,
    ROUND((n.quiz * 0.2 + n.uts * 0.3 + n.uas * 0.5), 2) AS nilai_akhir
FROM tb_nilai n
JOIN tb_mahasiswa m USING (nim)
JOIN tb_mk mk USING (kode_mk)
ORDER BY nilai_akhir DESC
LIMIT 1;  -- Ambil 1 baris teratas (nilai tertinggi)
```

---

### **3. Nilai Tertinggi per Mahasiswa**
Jika kamu ingin melihat **nilai tertinggi yang pernah dicapai setiap mahasiswa** (misal: UAS tertinggi per mahasiswa):

```sql
SELECT
    nim,
    MAX(uas) AS nilai_uas_tertinggi
FROM tb_nilai
GROUP BY nim;
```

> 💡 **Common Mistake:**
> Banyak yang lupa `GROUP BY nim` saat menggunakan `MAX()`. Tanpa `GROUP BY`, `MAX()` hanya akan mengembalikan **satu nilai tertinggi dari seluruh tabel**, bukan per mahasiswa.

---

### **4. Nilai Tertinggi per Mata Kuliah**
Jika kamu ingin melihat **nilai UAS tertinggi per mata kuliah**:

```sql
SELECT
    kode_mk,
    nama_mk,
    MAX(uas) AS nilai_uas_tertinggi
FROM tb_nilai n
JOIN tb_mk mk USING (kode_mk)
GROUP BY kode_mk, nama_mk;
```

## 12:02 · CODING
**Kamu:** libatkan mahasiswa dan nilainya saja

**Coding:** Berikut adalah query SQL yang **hanya melibatkan tabel `tb_mahasiswa` dan `tb_nilai`** untuk menentukan nilai tertinggi (baik UAS, UTS, quiz, atau nilai akhir) dari semua mahasiswa:

---

### **1. Nilai UAS Tertinggi (Hanya Mahasiswa & Nilai)**
Jika kamu ingin mencari **nilai UAS tertinggi** beserta **nama mahasiswa** yang mendapatkannya:

```sql
SELECT
    m.nim,
    m.nama,
    n.uas AS nilai_uas_tertinggi
FROM tb_nilai n
JOIN tb_mahasiswa m USING (nim)
WHERE n.uas = (SELECT MAX(uas) FROM tb_nilai);
```

**Penjelasan:**
- `SELECT MAX(uas) FROM tb_nilai`: Mencari nilai UAS tertinggi di seluruh tabel `tb_nilai`.
- `WHERE n.uas = (SELECT MAX(uas) FROM tb_nilai)`: Memfilter hanya baris yang memiliki nilai UAS sama dengan nilai tertinggi.
- `JOIN tb_mahasiswa m USING (nim)`: Menggabungkan dengan tabel `tb_mahasiswa` untuk mendapatkan nama mahasiswa.

---

### **2. Nilai Akhir Tertinggi (Gabungan Quiz, UTS, UAS)**
Jika kamu ingin mencari **nilai akhir tertinggi** (dihitung dari `quiz * 0.2 + uts * 0.3 + uas * 0.5`):

```sql
SELECT
    m.nim,
    m.nama,
    ROUND((n.quiz * 0.2 + n.uts * 0.3 + n.uas * 0.5), 2) AS nilai_akhir_tertinggi
FROM tb_nilai n
JOIN tb_mahasiswa m USING (nim)
ORDER BY nilai_akhir_tertinggi DESC
LIMIT 1;
```

**Penjelasan:**
- `ROUND((n.quiz * 0.2 + n.uts * 0.3 + n.uas * 0.5), 2)`: Menghitung nilai akhir dan membulatkannya menjadi 2 desimal.
- `ORDER BY nilai_akhir_tertinggi DESC`: Mengurutkan dari nilai tertinggi ke terendah.
- `LIMIT 1`: Mengambil hanya 1 baris teratas (nilai tertinggi).

---

### **3. Nilai Tertinggi per Mahasiswa**
Jika kamu ingin melihat **nilai tertinggi yang pernah dicapai setiap mahasiswa** (misal: UAS tertinggi per mahasiswa):

```sql
SELECT
    m.nim,
    m.nama,
    MAX(n.uas) AS nilai_uas_tertinggi
FROM tb_nilai n
JOIN tb_mahasiswa m USING (nim)
GROUP BY m.nim, m.nama;
```

**Penjelasan:**
- `GROUP BY m.nim, m.nama`: Mengelompokkan hasil berdasarkan mahasiswa.
- `MAX(n.uas)`: Mengambil nilai UAS tertinggi untuk setiap mahasiswa.

---

### **4. Nilai Tertinggi per Jenis Nilai (Quiz, UTS, UAS)**
Jika kamu ingin melihat **nilai tertinggi untuk setiap jenis nilai** (quiz, UTS, UAS):

```sql
SELECT
    'Quiz' AS jenis_nilai,
    MAX(n.quiz) AS nilai_tertinggi,
    m.nim,
    m.nama
FROM tb_nilai n
JOIN tb_mahasiswa m USING (nim)
WHERE n.quiz = (SELECT MAX(quiz) FROM tb_nilai)

UNION

SELECT
    'UTS' AS jenis_nilai,
    MAX(n.uts) AS nilai_tertinggi,
    m.nim,
    m.nama
FROM tb_nilai n
JOIN tb_mahasiswa m USING (nim)
WHERE n.uts = (SELECT MAX(uts) FROM tb_nilai)

UNION

SELECT
    'UAS' AS jenis_nilai,
    MAX(n.uas) AS nilai_tertinggi,
    m.nim,
    m.nama
FROM tb_nilai n
JOIN tb_mahasiswa m USING (nim)
WHERE n.uas = (SELECT MAX(uas) FROM tb_nilai);
```

**Penjelasan:**
- `UNION`: Menggabungkan hasil dari tiga query (quiz, UTS, UAS).
- Setiap query mencari nilai tertinggi untuk jenis nilai tertentu dan menampilkan mahasiswa yang mendapatkannya.
