function ymd(d) {
  const y = d.getFullYear();
  const m = String(d.getMonth()+1).padStart(2,'0');
  const day = String(d.getDate()).padStart(2,'0');
  return `${y}-${m}-${day}`;
}
function dLabel(d) { return `${d.getDate()} ${MONTHS_ID[d.getMonth()]}`; }

window.JOURNAL_DATA = (() => {
  const entries = RAW_ENTRIES.map(r => {
    const d = relDays(r.days);
    const wc = r.content.replace(/[#*>\-\n`]/g,' ').split(/\s+/).filter(Boolean).length;
    const firstLine = r.content.split('\n').filter(l => l.trim() && !l.startsWith('#'))[0] || '';
    return {
      date: ymd(d),
      date_label: dLabel(d),
      day_name: dayName(d),
      month_key: `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}`,
      month_label: `${MONTHS_ID[d.getMonth()]} ${d.getFullYear()}`,
      mood: r.mood,
      mood_cat: r.cat,
      content: r.content,
      preview: firstLine.replace(/[*_`]/g,'').slice(0, 200),
      word_count: wc,
      js_date: d,
    };
  });

  return {
    streak: 7,
    total_entries: entries.length,
    this_month_count: entries.filter(e => e.js_date.getMonth() === TODAY.getMonth()).length,
    today: entries[0] || null,
    entries,
    mood_history: entries.map(e => ({ date: e.date, mood_cat: e.mood_cat })),
    TODAY,
    DAYS_ID,
    MONTHS_ID,
  };
})();

window.JOURNAL_DATA.dLabel  = dLabel;
window.JOURNAL_DATA.dayName = dayName;
window.JOURNAL_DATA.ymd     = ymd;
